        const log = document.getElementById('debug-log');
        try {
            log.innerHTML += '<div>Initializing app...</div>';
            await window.app.init();
            log.innerHTML += '<div>App initialized. Navigating to settings...</div>';
            const container = document.getElementById('main-content');
            await window.app.renderSettingsModule(container);
            log.innerHTML += '<div>RenderSettingsModule finished successfully! HTML length: ' + container.innerHTML.length + '</div>';
            document.title = 'SETTINGS_OK_' + container.innerHTML.length;
        } catch (e) {
            log.innerHTML += '<div style="color:red;">ERROR: ' + e.message + ' | Stack: ' + e.stack + '</div>';
            document.title = 'SETTINGS_ERROR: ' + e.message;
        }
    });
