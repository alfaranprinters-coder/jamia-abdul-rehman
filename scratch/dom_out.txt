    <script>
        window.addEventListener('DOMContentLoaded', async () => {
            try {
                if (typeof MadrassahApp === 'undefined') {
                    document.title = 'APP_INIT_ERROR: MadrassahApp is undefined';
                    return;
                }
                const testApp = new MadrassahApp();
                await testApp.init();
                const container = document.getElementById('main-content');
                await testApp.renderSettingsModule(container);
                if (container.innerHTML.includes('selectCustomBackupFolder') &&
                    container.innerHTML.includes('updateAutoBackupInterval')) {
                    document.title = 'APP_INIT_SUCCESS: SETTINGS_RENDERED_OK';
                } else {
                    document.title = 'APP_INIT_ERROR: Missing elements in settings';
                }
            } catch (err) {
                console.error('ERROR: ' + err.message);
                document.title = 'APP_INIT_ERROR: ' + err.message;
            }
        });
    </script>
